      function determineWinnerFromPosition() {
        const winningZoneRect = winningZone.getBoundingClientRect();
        const zoneCenterX = winningZoneRect.left + winningZoneRect.width / 2;
        const zoneCenterY = winningZoneRect.top + winningZoneRect.height / 2;

        const cards = Array.from(prizeTrack.querySelectorAll('.prize-card'));
        let closestCard = null;
        let minDistance = Infinity;

        cards.forEach(card => {
          const rect = card.getBoundingClientRect();
          const cardCenterX = rect.left + rect.width / 2;
          const cardCenterY = rect.top + rect.height / 2;
          const dx = cardCenterX - zoneCenterX;
          const dy = cardCenterY - zoneCenterY;
          const dist = Math.sqrt(dx * dx + dy * dy);
          if (dist < minDistance) {
            minDistance = dist;
            closestCard = card;
          }
        });

        if (!closestCard) return null;
        const prizeId = parseInt(closestCard.dataset.prizeId, 10);
        const prize = prizes.find(p => p.id === prizeId) || prizes[0];
        return { cardElement: closestCard, prize, distance: minDistance };
      }
